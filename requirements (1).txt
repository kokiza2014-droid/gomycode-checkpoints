from __future__ import annotations

import io
import sqlite3
from datetime import date
from pathlib import Path

import pandas as pd
import plotly.express as px
import streamlit as st

APP_DIR = Path(__file__).resolve().parent
DB_FILE = APP_DIR / "simbox_dashboard.db"

st.set_page_config(page_title="SIM Box Dashboard", page_icon="📡", layout="wide")

DEFAULT_SETTINGS = {
    "simbox_cost_per_minute": 5.0,
    "retail_rate": 8.0,
    "onnet_wholesale_rate_usd": 0.56,
    "offnet_wholesale_rate_usd": 0.45,
    "national_interco_rate_onnet": 0.68,
    "national_interco_rate_offnet": 0.54,
}


def connect():
    con = sqlite3.connect(DB_FILE)
    con.row_factory = sqlite3.Row
    return con


def init_db():
    with connect() as con:
        con.execute("""
            CREATE TABLE IF NOT EXISTS monthly_results (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                record_date TEXT NOT NULL UNIQUE,
                usd_dzd REAL NOT NULL,
                ooredoo_suspensions INTEGER DEFAULT 0,
                ooredoo_minutes REAL DEFAULT 0,
                ooredoo_calls INTEGER DEFAULT 0,
                preventive_suspensions INTEGER DEFAULT 0,
                djezzy_minutes REAL DEFAULT 0,
                mobilis_from_ooredoo_minutes REAL DEFAULT 0,
                fixed_minutes REAL DEFAULT 0,
                loss_averted_usd REAL DEFAULT 0,
                mobilis_distinct INTEGER DEFAULT 0,
                mobilis_duration REAL DEFAULT 0,
                mobilis_calls INTEGER DEFAULT 0,
                ota_distinct INTEGER DEFAULT 0,
                ota_duration REAL DEFAULT 0,
                ota_calls INTEGER DEFAULT 0,
                fixed_distinct INTEGER DEFAULT 0,
                mou REAL DEFAULT 0,
                ooredoo_total_minutes REAL DEFAULT 0,
                onnet_risk_usd REAL DEFAULT 0,
                mobilis_risk_usd REAL DEFAULT 0,
                ota_risk_usd REAL DEFAULT 0,
                total_risk_usd REAL DEFAULT 0,
                offnet_minutes REAL DEFAULT 0,
                total_minutes REAL DEFAULT 0,
                notes TEXT
            )
        """)
        con.execute("""
            CREATE TABLE IF NOT EXISTS settings (
                setting_key TEXT PRIMARY KEY,
                setting_value REAL NOT NULL
            )
        """)
        for key, value in DEFAULT_SETTINGS.items():
            con.execute(
                "INSERT OR IGNORE INTO settings VALUES (?, ?)",
                (key, value),
            )


def settings_load():
    with connect() as con:
        rows = con.execute("SELECT * FROM settings").fetchall()
    return {r["setting_key"]: float(r["setting_value"]) for r in rows}


def settings_save(values):
    with connect() as con:
        for key, value in values.items():
            con.execute("""
                INSERT INTO settings(setting_key, setting_value)
                VALUES (?, ?)
                ON CONFLICT(setting_key)
                DO UPDATE SET setting_value=excluded.setting_value
            """, (key, float(value)))


def data_load():
    with connect() as con:
        return pd.read_sql_query(
            "SELECT * FROM monthly_results ORDER BY record_date", con
        )


def data_save(data):
    columns = list(data)
    placeholders = ",".join("?" for _ in columns)
    updates = ",".join(
        f"{c}=excluded.{c}" for c in columns if c != "record_date"
    )
    sql = f"""
        INSERT INTO monthly_results ({','.join(columns)})
        VALUES ({placeholders})
        ON CONFLICT(record_date) DO UPDATE SET {updates}
    """
    with connect() as con:
        con.execute(sql, [data[c] for c in columns])


def safe_div(a, b):
    return a / b if b else 0.0


def calc_ooredoo(v, s):
    offnet_duration = v["djezzy"] + v["mobilis"] + v["fixed"]
    total_minutes = v["ooredoo"] + offnet_duration
    mou = safe_div(total_minutes, v["calls"])
    spend_dzd = total_minutes * s["simbox_cost_per_minute"]
    expected_dzd = (
        s["onnet_wholesale_rate_usd"] * v["usd_dzd"] * v["ooredoo"]
    )
    onnet_loss_usd = safe_div(expected_dzd - spend_dzd, v["usd_dzd"])
    offnet_margin_dzd = offnet_duration * (
        s["retail_rate"] - s["national_interco_rate_onnet"]
    )
    offnet_loss_usd = safe_div(offnet_margin_dzd, v["usd_dzd"])
    return {
        "offnet_duration": offnet_duration,
        "total_minutes": total_minutes,
        "mou": mou,
        "risk_usd": onnet_loss_usd + offnet_loss_usd + v["loss_averted"],
    }


def calc_offnet(v, s):
    wholesale_dzd = s["offnet_wholesale_rate_usd"] * v["usd_dzd"]
    loss_dzd = (
        wholesale_dzd - s["national_interco_rate_offnet"]
    ) * v["duration"]
    return {
        "duration_per_sim": safe_div(v["duration"], v["distinct"]),
        "risk_usd": safe_div(loss_dzd, v["usd_dzd"]),
    }


def excel_export(df):
    out = io.BytesIO()
    export = df.copy()
    export["distinct_offnet"] = (
        export["mobilis_distinct"] + export["ota_distinct"] + export["fixed_distinct"]
    )
    export["offnet_risk_usd"] = export["mobilis_risk_usd"] + export["ota_risk_usd"]
    with pd.ExcelWriter(out, engine="openpyxl") as writer:
        export.to_excel(writer, sheet_name="Donnees consolidees", index=False)
        export[[
            "record_date", "ooredoo_suspensions", "distinct_offnet",
            "ota_distinct", "mobilis_distinct", "fixed_distinct",
            "total_risk_usd", "onnet_risk_usd", "offnet_risk_usd",
            "total_minutes", "ooredoo_total_minutes", "ota_duration",
            "mobilis_duration", "offnet_minutes"
        ]].to_excel(writer, sheet_name="Summary", index=False)
    return out.getvalue()


def read_sheet(uploaded, names, header=0):
    for name in names:
        try:
            uploaded.seek(0)
            return pd.read_excel(uploaded, sheet_name=name, header=header)
        except ValueError:
            pass
    raise ValueError(f"Feuille introuvable: {names}")


def n(row, column):
    value = row.get(column, 0) if hasattr(row, "get") else 0
    return 0.0 if pd.isna(value) else float(value)


def import_excel(uploaded):
    s = settings_load()
    odf = read_sheet(uploaded, ["Ooredoo SB Loss"], header=1)
    mdf = read_sheet(uploaded, ["Mobilis SB Loss"])
    tdf = read_sheet(uploaded, ["OTA SB Loss ", "OTA SB Loss"])
    count = 0
    for _, r in odf.iterrows():
        if pd.isna(r.get("Date")):
            continue
        month = pd.to_datetime(r["Date"]).strftime("%Y-%m")
        usd = n(r, "USD - DZD ")
        mmask = pd.to_datetime(mdf["Date"], errors="coerce").dt.strftime("%Y-%m") == month
        tmask = pd.to_datetime(tdf["Date"], errors="coerce").dt.strftime("%Y-%m") == month
        mr = mdf[mmask].iloc[0] if mmask.any() else {}
        tr = tdf[tmask].iloc[0] if tmask.any() else {}

        ov = {
            "usd_dzd": usd,
            "ooredoo": n(r, "terminated to Ooredoo  (min)"),
            "calls": int(n(r, "Number of calls")),
            "djezzy": n(r, "terminated to Djezzy   (min)"),
            "mobilis": n(r, "terminated to Mobilis    (min)"),
            "fixed": n(r, "terminated to Fixe        (min)"),
            "loss_averted": n(r, "Loss Averted"),
        }
        mv = {
            "usd_dzd": usd,
            "distinct": int(n(mr, "Number of distinct SB-Mobilis")),
            "duration": n(mr, "Mobilis Terminated Duration"),
        }
        tv = {
            "usd_dzd": usd,
            "distinct": int(n(tr, "Number of distinct SB-OTA")),
            "duration": n(tr, "OTA Terminated Duration"),
        }
        oc, mc, tc = calc_ooredoo(ov, s), calc_offnet(mv, s), calc_offnet(tv, s)
        offnet_minutes = mv["duration"] + tv["duration"]
        data_save({
            "record_date": month,
            "usd_dzd": usd,
            "ooredoo_suspensions": int(n(r, "Number of Suspension")),
            "ooredoo_minutes": ov["ooredoo"],
            "ooredoo_calls": ov["calls"],
            "preventive_suspensions": int(n(r, "Number of Suspension preventives")),
            "djezzy_minutes": ov["djezzy"],
            "mobilis_from_ooredoo_minutes": ov["mobilis"],
            "fixed_minutes": ov["fixed"],
            "loss_averted_usd": ov["loss_averted"],
            "mobilis_distinct": mv["distinct"],
            "mobilis_duration": mv["duration"],
            "mobilis_calls": int(n(mr, "Mobilis-Number of calls")),
            "ota_distinct": tv["distinct"],
            "ota_duration": tv["duration"],
            "ota_calls": int(n(tr, "OTA-Number of calls")),
            "fixed_distinct": 0,
            "mou": oc["mou"],
            "ooredoo_total_minutes": oc["total_minutes"],
            "onnet_risk_usd": oc["risk_usd"],
            "mobilis_risk_usd": mc["risk_usd"],
            "ota_risk_usd": tc["risk_usd"],
            "total_risk_usd": oc["risk_usd"] + mc["risk_usd"] + tc["risk_usd"],
            "offnet_minutes": offnet_minutes,
            "total_minutes": oc["total_minutes"] + offnet_minutes,
            "notes": "Importe depuis le fichier historique",
        })
        count += 1
    return count
